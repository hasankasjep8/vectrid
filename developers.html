Quick Start

Get vectorizing in under 5 lines of code.

# Install: pip install vectrid
import vectrid

client = vectrid.Client("your_api_key")

# Convert an image to SVG
result = client.vectorize("photo.png")
result.save("output.svg")

# With options
result = client.vectorize(
    "logo.jpg",
    mode="spline",
    colors=16,
    output_format="svg"
)
